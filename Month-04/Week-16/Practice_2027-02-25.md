## ১. রিভিশন (২০ মিনিট)

খাতা না দেখে উত্তর দাও। না পারলে সেই ক্লাসের ফাইল আবার পড়ো।

**ক্লাস ৭৬:** `UPDATE` কোনো রো না পেলে কী হয়? `SQL%ROWCOUNT` কোন SQL-এর কথা বলে? `SELECT INTO` আর `UPDATE`-এর ০ রো-তে পার্থক্য কী?

**ক্লাস ৭৭:** Explicit Cursor-এর চার ধাপ কী কী? শেষ রো-র পরে `FETCH` করলে কী হয়?

**ক্লাস ৭৮:** কোন Attribute বন্ধ Cursor-এ এরর দেয় না? প্রথম `FETCH`-এর আগে `%NOTFOUND` কী? `WHILE %FOUND`-এর আগে কী লাগে?

**ক্লাস ৭৯:** Cursor FOR Oracle-কে দিয়ে কোন কাজগুলো করায়? Loop-এর পরে মোট রো সংখ্যা কীভাবে জানবে? Control Break-এ কোন অংশ ভুলে যায়?

**ক্লাস ৮০:** Cursor Parameter-এর ধরনে কী দেওয়া যায় না? `WHERE CURRENT OF`-এর জন্য কী লাগে? `FOR UPDATE` Loop-এ `COMMIT` কোথায়?

## ২. মূল কাজ (১১৫ মিনিট)

`learn_freepdb1`-এ করো। প্রতিটা প্রোগ্রাম `week16.sql`-এ কমেন্টে নম্বর দিয়ে রাখো।

**আজকের গল্প:** তুমি একটা গার্মেন্টস ফ্যাক্টরির ERP টিমে। HR জানুয়ারি ২০২৭-এর অনুপস্থিতির হিসাব দিয়েছে। মাসের স্যালারি শিট বানাতে হবে, তারপর হিসাব বিভাগের জন্য রিপোর্ট।

**স্যালারির নিয়ম:**

| অংশ | হিসাব |
|---|---|
| Basic | `w16_emp.basic_salary` |
| বাড়িভাড়া (House Rent) | Basic-এর ৫০% |
| চিকিৎসা (Medical) | ৭৫০ |
| যাতায়াত (Transport) | ৪৫০ |
| Gross | Basic + বাড়িভাড়া + চিকিৎসা + যাতায়াত |
| অনুপস্থিতির কাটা | `ROUND(Basic / 30 × অনুপস্থিত দিন)` |
| Net | Gross − কাটা |

শুধু `ACTIVE` কর্মচারীর স্যালারি হবে।

**নিয়ম:**
- ৭৫০, ৪৫০, ০.৫ এসব Constant-এ রাখো, সরাসরি সংখ্যা ছড়িয়ে লিখো না
- Cursor-এ সবসময় `ORDER BY`
- টেবিল বদলালে পরীক্ষার পরে `ROLLBACK`, যদি না বলা থাকে

`a.absent_month = '2027-01'` শর্ত `ON`-এ, `WHERE`-এ না। `WHERE`-এ দিলে যাদের অনুপস্থিতি নেই তারা বাদ পড়ত (ক্লাস ৪২-এর `ON` নাকি `WHERE`)।

```sql
-- ৩ ও ৪
DECLARE
    c_month      CONSTANT VARCHAR2(7) := '2027-01';
    c_house_pct  CONSTANT NUMBER := 0.5;
    c_medical    CONSTANT NUMBER := 750;
    c_transport  CONSTANT NUMBER := 450;
    c_month_days CONSTANT NUMBER := 30;

    CURSOR c_emp (p_month VARCHAR2) IS
        SELECT e.emp_id, e.basic_salary, NVL(a.absent_days, 0) AS absent_days
        FROM   w16_emp e
               LEFT JOIN w16_absent a
                      ON a.emp_id = e.emp_id
                     AND a.absent_month = p_month
        WHERE  e.status = 'ACTIVE'
        ORDER  BY e.emp_id;

    v_house  NUMBER;
    v_gross  NUMBER;
    v_ded    NUMBER;
    v_cnt    NUMBER := 0;
    v_total  NUMBER := 0;
BEGIN
    -- ৪: আবার চালানো নিরাপদ করা
    DELETE FROM w16_salary WHERE salary_month = c_month;
    DBMS_OUTPUT.PUT_LINE('Old rows removed: ' || SQL%ROWCOUNT);

    FOR r IN c_emp(c_month) LOOP
        v_house := r.basic_salary * c_house_pct;
        v_gross := r.basic_salary + v_house + c_medical + c_transport;
        v_ded   := ROUND(r.basic_salary / c_month_days * r.absent_days);

        INSERT INTO w16_salary
               (salary_month, emp_id, basic_salary, house_rent, medical, transport,
                gross_salary, absent_days, deduction, net_salary)
        VALUES (c_month, r.emp_id, r.basic_salary, v_house, c_medical, c_transport,
                v_gross, r.absent_days, v_ded, v_gross - v_ded);

        v_cnt   := v_cnt + 1;
        v_total := v_total + (v_gross - v_ded);
    END LOOP;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Rows: ' || v_cnt || ', Total net: ' || v_total);   -- 9, 176090
END;
/

-- ৫
SELECT SUM(gross_salary), SUM(deduction), SUM(net_salary)

-- ৬

```

`DELETE` আর সব `INSERT` একই Transaction-এ, শেষে একটা `COMMIT`। মাঝপথে কিছু ভাঙলে পুরনো শিট হারায় না। ERP-তে "Re-process" বাটন এভাবেই কাজ করে।

-- ৭
DECLARE
    CURSOR c_sheet (p_month VARCHAR2) IS
        SELECT e.emp_name, s.basic_salary, s.house_rent, s.gross_salary,
               s.deduction, s.net_salary
        FROM   w16_salary s

        ORDER  BY s.emp_id;

    v_gross NUMBER := 0;
    v_ded   NUMBER := 0;
    v_net   NUMBER := 0;

    DBMS_OUTPUT.PUT_LINE('Salary Sheet: January 2027');

    FOR r IN c_sheet('2027-01') LOOP
        DBMS_OUTPUT.PUT_LINE(RPAD(c_sheet%ROWCOUNT, 4) || RPAD(r.emp_name, 10)
                             || LPAD(r.basic_salary, 8) || LPAD(r.house_rent, 8)
                             || LPAD(r.gross_salary, 8) || LPAD(r.deduction, 6)
                             || LPAD(r.net_salary, 8));
        v_gross := v_gross + r.gross_salary;
        v_ded   := v_ded + r.deduction;
        v_net   := v_net + r.net_salary;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE(RPAD('-', 52, '-'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Total', 30) || LPAD(v_gross, 8) || LPAD(v_ded, 6) || LPAD(v_net, 8));

-- ৮
DECLARE
    v_emp_id NUMBER := 3;               -- তারপর 1, 99
    v_month  VARCHAR2(7) := '2027-01';
    v_found  BOOLEAN := FALSE;

    CURSOR c_slip (p_emp_id NUMBER, p_month VARCHAR2) IS
        SELECT e.emp_name, e.dept_name, s.*
        FROM   w16_salary s
               JOIN w16_emp e ON e.emp_id = s.emp_id
        WHERE  s.emp_id = p_emp_id
        AND    s.salary_month = p_month;
BEGIN
    FOR r IN c_slip(v_emp_id, v_month) LOOP
        v_found := TRUE;
        DBMS_OUTPUT.PUT_LINE('PAYSLIP ' || r.salary_month);
        DBMS_OUTPUT.PUT_LINE(r.emp_name || ' (' || r.dept_name || ')');
        DBMS_OUTPUT.PUT_LINE(RPAD('Basic', 14)      || LPAD(r.basic_salary, 8));
        DBMS_OUTPUT.PUT_LINE(RPAD('House Rent', 14) || LPAD(r.house_rent, 8));
        DBMS_OUTPUT.PUT_LINE(RPAD('Medical', 14)    || LPAD(r.medical, 8));
        DBMS_OUTPUT.PUT_LINE(RPAD('Transport', 14)  || LPAD(r.transport, 8));
        DBMS_OUTPUT.PUT_LINE(RPAD('Gross', 14)      || LPAD(r.gross_salary, 8));
        DBMS_OUTPUT.PUT_LINE(RPAD('Absent (' || r.absent_days || 'd)', 14) || LPAD('-' || r.deduction, 8));
        DBMS_OUTPUT.PUT_LINE(RPAD('NET PAY', 14)    || LPAD(r.net_salary, 8));
    END LOOP;

    IF NOT v_found THEN
        DBMS_OUTPUT.PUT_LINE('No salary for employee ' || v_emp_id || ' in ' || v_month);
    END IF;
END;
