-- ১৬
DECLARE
    v_item_id w14_stock.item_id%TYPE := 1;
    v_qty     w14_txn.qty%TYPE := 100;
    v_balance w14_stock.balance_qty%TYPE;
    v_count   NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count FROM w14_stock WHERE item_id = v_item_id;

    IF v_count = 0 THEN
        DBMS_OUTPUT.PUT_LINE('Item not found');
    ELSE
        SELECT balance_qty INTO v_balance FROM w14_stock WHERE item_id = v_item_id;

        IF v_qty > v_balance THEN
            DBMS_OUTPUT.PUT_LINE('Short by ' || (v_qty - v_balance));
        ELSE
            INSERT INTO w14_txn (item_id, txn_type, qty) VALUES (v_item_id, 'ISSUE', v_qty);

            UPDATE w14_stock
            SET    balance_qty = balance_qty - v_qty
            WHERE  item_id = v_item_id
            RETURNING balance_qty INTO v_balance;

            DBMS_OUTPUT.PUT_LINE('New balance: ' || v_balance);
        END IF;
    END IF;

    ROLLBACK;
END;
/

-- ১৭
DECLARE
    v_member_id lib_member.member_id%TYPE := 1;
    v_copy_id   lib_book_copy.copy_id%TYPE := 1;
    v_count     NUMBER;
    v_active    lib_member.is_active%TYPE;
    v_max_books lib_member_type.max_books%TYPE;
    v_status    lib_book_copy.copy_status%TYPE;
    v_out       NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count FROM lib_member WHERE member_id = v_member_id;

    IF v_count = 0 THEN
        DBMS_OUTPUT.PUT_LINE('Member not found');
    ELSE
        SELECT m.is_active, t.max_books
        INTO   v_active, v_max_books
        FROM   lib_member m
        JOIN   lib_member_type t ON m.member_type_id = t.member_type_id
        WHERE  m.member_id = v_member_id;

        SELECT copy_status INTO v_status FROM lib_book_copy WHERE copy_id = v_copy_id;

        SELECT COUNT(*) INTO v_out
        FROM   lib_book_issue
        WHERE  member_id = v_member_id
        AND    return_date IS NULL;

        IF v_active <> 'Y' THEN
            DBMS_OUTPUT.PUT_LINE('Member is inactive');
        ELSIF v_status <> 'AVAILABLE' THEN
            DBMS_OUTPUT.PUT_LINE('Not available: ' || v_status);
        ELSIF v_out >= v_max_books THEN
            DBMS_OUTPUT.PUT_LINE('Limit reached');
        ELSE
            DBMS_OUTPUT.PUT_LINE('Can issue');
        END IF;
    END IF;
END;
/
