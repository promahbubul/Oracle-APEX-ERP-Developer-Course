## ২. মূল কাজ (১১৫ মিনিট)

`learn_freepdb1`-এ করো। Spec আর Body `week19.sql`-এ, শুরুতে ক্লাস ৯৫-এর চেকলিস্ট।

**আজকের লক্ষ্য:** সপ্তাহ ১৬ থেকে ১৮ পর্যন্ত ছড়ানো সব স্যালারির কোড এক Package-এ: `w19_payroll_pkg`। নিয়ম আসবে `w19_setting_pkg` থেকে (ক্লাস ৯৪), লগ আর যাচাই `w19_util_pkg` থেকে (ক্লাস ৯৫), নাম `w19_hr_pkg` থেকে (ক্লাস ৯১)।

```
                     w19_payroll_pkg
        ┌───────────────┼────────────────┐
  w19_setting_pkg   w19_hr_pkg      w19_util_pkg
   (নিয়ম, Cache)    (emp_name)    (assert, log, month_label)
```

শুরুর আগে মিলিয়ে নাও: `w19_payroll_setting`-এ ৫টা রো আর মান আগের মতো (চিকিৎসা ৭৫০), `w16_salary`-তে জানুয়ারির ৯টা রো, আর তিনটা সাহায্যকারী Package VALID।

### Spec-এ যা থাকবে

| ধরন | নাম | কাজ |
|---|---|---|
| Constant | `c_err_no_emp` … `c_err_bad_month` | `-20201` থেকে `-20207` |
| Exception | `e_emp_left`, `e_below_wage` | `PRAGMA` সহ |
| Function | `gross(p_basic)` | সেটিং থেকে Gross |
| Function | `deduction(p_basic, p_days)` | অনুপস্থিতির কাটা |
| Function | `net(p_basic, p_days)` | Gross − কাটা |
| Procedure | `check_emp(p_emp_id)` | না থাকলে / `LEFT` / ন্যূনতম মজুরির নিচে: এরর |
| Procedure | `process_emp(p_month, p_emp_id)` | একজনের রো মুছে নতুন করে ঢোকানো |
| Procedure | `process_month(p_month, p_dept DEFAULT NULL)` | সবার (বা একটা বিভাগের), Skip গোনা |
| Function | `last_processed`, `last_skipped` | শেষ `process_month`-এর ফল (Package Variable) |
| Procedure | `payslip(p_emp_id NUMBER, p_month)` | একজনের Payslip |
| Procedure | `payslip(p_emp_name VARCHAR2, p_month)` | নাম দিয়ে (Overload) |
| Procedure | `dept_report(p_month)` | বিভাগভিত্তিক Control Break |
| Function | `month_total(p_month, p_dept DEFAULT NULL)` | মোট Net, SQL থেকেও |

| এরর | অর্থ |
|---|---|
| -20201 | কর্মচারী নেই |
| -20202 | ন্যূনতম মজুরির নিচে |

### Body-তে Private

| নাম | কাজ |
|---|---|
| `validate_month(p_month)` | `TO_DATE(p_month || '-01', 'YYYY-MM-DD')` চেষ্টা, না হলে `-20207`; আর দৈর্ঘ্য ঠিক ৭ |

### কাজের তালিকা

1. সাহায্যকারী তিন Package (`w19_setting_pkg`, `w19_hr_pkg`, `w19_util_pkg`) VALID কি না দেখো, দরকার হলে আবার Compile।
2. `w19_payroll_pkg`-এর Spec লেখো (ক্লাস ৯৫ প্র্যাকটিস ১০ থেকে), কমেন্টসহ। Compile।
3. Body: Private অংশ আর তিনটা হিসাবের Function। SQL-এ মিলাও: `SUM(w19_payroll_pkg.gross(basic_salary))` (১৭৮৮০০), Salma-র `net(11000, 2)` (১৬৯৬৭)।
4. `check_emp`: `w19_setting_pkg.min_wage` দিয়ে। ১, ১০, ৯৯ দিয়ে।
5. `process_emp`: ক্লাস ৮৯-এর মতো, কিন্তু সব সংখ্যা সেটিং থেকে, আর শুরুতে `validate_month`। `('2027-01', 3)`, `('2027-1', 3)`, `('2027-13', 3)` দিয়ে।
6. `process_month`: শুরুতে `validate_month` আর `w19_setting_pkg.reload`, তারপর ক্লাস ৯০-এর মতো Loop। `e_emp_left` আর `e_below_wage` ধরে Skip, বাকি এরর উপরে। শেষে `g_last_processed`/`g_last_skipped` বসাও। `COMMIT` নেই।
7. `process_month('2027-01')` চালাও, তারপর `last_processed`, `last_skipped` আর `month_total('2027-01')`। (৯, ১, ১৭৬০৯০) `COMMIT`।
8. `process_month('2027-01', 'Sewing')`, তারপর `'Store'`। (৩/০; ০/১) `month_total('2027-01', 'Sewing')`? (৫৪৬১৭)

CREATE OR REPLACE PACKAGE BODY w19_payroll_pkg
IS
    g_last_processed NUMBER := 0;
    g_last_skipped   NUMBER := 0;

    -- ======================== Private ========================

    PROCEDURE validate_month (p_month IN VARCHAR2)
    IS
        v_dummy DATE;
    BEGIN
        w19_util_pkg.assert(LENGTH(p_month) = 7, c_err_bad_month,
                            'Month must be YYYY-MM, got ' || NVL(p_month, 'NULL'));
        BEGIN
            v_dummy := TO_DATE(p_month || '-01', 'YYYY-MM-DD');
        EXCEPTION
            WHEN OTHERS THEN
                RAISE_APPLICATION_ERROR(c_err_bad_month, 'Month must be YYYY-MM, got ' || p_month);
        END;
    END validate_month;

    FUNCTION absent_days (
        p_emp_id IN w16_emp.emp_id%TYPE,
        p_month  IN VARCHAR2
    ) RETURN NUMBER
    IS
        v_days NUMBER;
    BEGIN
        SELECT NVL(MAX(absent_days), 0) INTO v_days
        FROM   w16_absent
        WHERE  emp_id = p_emp_id AND absent_month = p_month;
        RETURN v_days;
    END absent_days;

    -- ======================== হিসাব ========================

    FUNCTION gross (p_basic IN NUMBER) RETURN NUMBER
    IS
    BEGIN
        RETURN p_basic
             + p_basic * w19_setting_pkg.house_pct
             + w19_setting_pkg.medical
             + w19_setting_pkg.transport;
    END gross;

    FUNCTION deduction (p_basic IN NUMBER, p_days IN NUMBER) RETURN NUMBER
    IS
    BEGIN
        RETURN ROUND(p_basic / w19_setting_pkg.month_days * NVL(p_days, 0));
    END deduction;

    FUNCTION net (p_basic IN NUMBER, p_days IN NUMBER) RETURN NUMBER
    IS
    BEGIN
        RETURN gross(p_basic) - deduction(p_basic, p_days);
    END net;

    -- ======================== প্রসেস ========================

    PROCEDURE check_emp (p_emp_id IN w16_emp.emp_id%TYPE)
    IS
        v_status w16_emp.status%TYPE;
        v_basic  w16_emp.basic_salary%TYPE;
    BEGIN
        BEGIN
            SELECT status, basic_salary INTO v_status, v_basic
            FROM   w16_emp WHERE emp_id = p_emp_id;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(c_err_no_emp, 'Employee ' || p_emp_id || ' not found');
        END;

        w19_util_pkg.assert(v_status <> 'LEFT', c_err_emp_left,
                            w19_hr_pkg.emp_name(p_emp_id) || ' has left, no salary');
        w19_util_pkg.assert(v_basic >= w19_setting_pkg.min_wage, c_err_below_wage,
                            w19_hr_pkg.emp_name(p_emp_id) || ': basic ' || v_basic
                            || ' below minimum wage ' || w19_setting_pkg.min_wage);
    END check_emp;

    PROCEDURE process_emp (
        p_month  IN w16_salary.salary_month%TYPE,
        p_emp_id IN w16_emp.emp_id%TYPE
    )
    IS
        v_basic w16_emp.basic_salary%TYPE;
        v_days  NUMBER;
        v_house NUMBER;
        v_gross NUMBER;
        v_ded   NUMBER;
    BEGIN
        validate_month(p_month);
        check_emp(p_emp_id);

        SELECT basic_salary INTO v_basic FROM w16_emp WHERE emp_id = p_emp_id;
        v_days  := absent_days(p_emp_id, p_month);
        v_house := v_basic * w19_setting_pkg.house_pct;
        v_gross := gross(v_basic);
        v_ded   := deduction(v_basic, v_days);

        DELETE FROM w16_salary WHERE salary_month = p_month AND emp_id = p_emp_id;

        INSERT INTO w16_salary
               (salary_month, emp_id, basic_salary, house_rent, medical, transport,
                gross_salary, absent_days, deduction, net_salary)
        VALUES (p_month, p_emp_id, v_basic, v_house,
                w19_setting_pkg.medical, w19_setting_pkg.transport,
                v_gross, v_days, v_ded, v_gross - v_ded);
    END process_emp;

    PROCEDURE process_month (
        p_month IN w16_salary.salary_month%TYPE,
        p_dept  IN w16_emp.dept_name%TYPE DEFAULT NULL
    )
    IS
        v_ok   NUMBER := 0;
        v_skip NUMBER := 0;
    BEGIN
        validate_month(p_month);
        w19_setting_pkg.reload;                -- সর্বশেষ নিয়ম

        FOR r IN (SELECT emp_id FROM w16_emp
                  WHERE  (dept_name = p_dept OR p_dept IS NULL)
                  ORDER  BY emp_id)
        LOOP
            BEGIN
                process_emp(p_month, r.emp_id);
                v_ok := v_ok + 1;
            EXCEPTION
                WHEN e_emp_left OR e_below_wage THEN
                    DBMS_OUTPUT.PUT_LINE('Skipped: ' || SQLERRM);
                    v_skip := v_skip + 1;
            END;
        END LOOP;

        g_last_processed := v_ok;
        g_last_skipped   := v_skip;
        DBMS_OUTPUT.PUT_LINE(w19_util_pkg.month_label(p_month)
                             || NVL2(p_dept, ' (' || p_dept || ')', '')
                             || ': ' || v_ok || ' processed, ' || v_skip || ' skipped');
    END process_month;

    FUNCTION last_processed RETURN NUMBER IS BEGIN RETURN g_last_processed; END last_processed;
    FUNCTION last_skipped   RETURN NUMBER IS BEGIN RETURN g_last_skipped;   END last_skipped;

    -- ======================== রিপোর্ট ========================

    PROCEDURE payslip (p_emp_id IN NUMBER, p_month IN VARCHAR2)
    IS
        r w16_salary%ROWTYPE;
    BEGIN
        validate_month(p_month);
        BEGIN
            SELECT * INTO r FROM w16_salary WHERE emp_id = p_emp_id AND salary_month = p_month;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(c_err_no_salary,
                    'No salary for employee ' || p_emp_id || ' in ' || p_month);
        END;

        DBMS_OUTPUT.PUT_LINE('PAYSLIP - ' || w19_util_pkg.month_label(p_month));
        DBMS_OUTPUT.PUT_LINE(w19_hr_pkg.emp_name(p_emp_id) || ' (ID ' || p_emp_id || ')');
        DBMS_OUTPUT.PUT_LINE(RPAD('Basic', 16)      || LPAD(r.basic_salary, 9));
        DBMS_OUTPUT.PUT_LINE(RPAD('House Rent', 16) || LPAD(r.house_rent, 9));
        DBMS_OUTPUT.PUT_LINE(RPAD('Medical', 16)    || LPAD(r.medical, 9));
        DBMS_OUTPUT.PUT_LINE(RPAD('Transport', 16)  || LPAD(r.transport, 9));
        DBMS_OUTPUT.PUT_LINE(RPAD('Gross', 16)      || LPAD(r.gross_salary, 9));
        DBMS_OUTPUT.PUT_LINE(RPAD('Absent ' || r.absent_days || 'd', 16) || LPAD('-' || r.deduction, 9));
        DBMS_OUTPUT.PUT_LINE(RPAD('NET PAY', 16)    || LPAD(r.net_salary, 9));
    END payslip;

    PROCEDURE payslip (p_emp_name IN VARCHAR2, p_month IN VARCHAR2)
    IS
        v_id w16_emp.emp_id%TYPE;
    BEGIN
        BEGIN
            SELECT emp_id INTO v_id FROM w16_emp WHERE UPPER(emp_name) = UPPER(TRIM(p_emp_name));
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(c_err_no_emp, 'No employee named ' || p_emp_name);
            WHEN TOO_MANY_ROWS THEN
                RAISE_APPLICATION_ERROR(c_err_many_names, 'Several employees named ' || p_emp_name || ', use ID');
        END;
        payslip(v_id, p_month);
    END payslip;

    PROCEDURE dept_report (p_month IN w16_salary.salary_month%TYPE)
    IS
        v_prev  w16_emp.dept_name%TYPE;
        v_sub   NUMBER := 0;
        v_grand NUMBER := 0;
    BEGIN
        validate_month(p_month);
        DBMS_OUTPUT.PUT_LINE('Department salary report - ' || w19_util_pkg.month_label(p_month));

        FOR r IN (SELECT e.dept_name, e.emp_name, s.net_salary
                  FROM   w16_salary s JOIN w16_emp e ON e.emp_id = s.emp_id
                  WHERE  s.salary_month = p_month
                  ORDER  BY e.dept_name, s.emp_id)
        LOOP
            IF v_prev IS NOT NULL AND r.dept_name <> v_prev THEN
                DBMS_OUTPUT.PUT_LINE('  Subtotal ' || RPAD(v_prev, 10) || LPAD(v_sub, 9));
                v_sub := 0;
            END IF;
            IF v_prev IS NULL OR r.dept_name <> v_prev THEN
                DBMS_OUTPUT.PUT_LINE('== ' || r.dept_name);
            END IF;
            DBMS_OUTPUT.PUT_LINE('    ' || RPAD(r.emp_name, 16) || LPAD(r.net_salary, 9));
            v_sub   := v_sub + r.net_salary;
            v_grand := v_grand + r.net_salary;
            v_prev  := r.dept_name;
        END LOOP;

        IF v_prev IS NULL THEN
            RAISE_APPLICATION_ERROR(c_err_no_salary, 'Salary for ' || p_month || ' not processed');
        END IF;
        DBMS_OUTPUT.PUT_LINE('  Subtotal ' || RPAD(v_prev, 10) || LPAD(v_sub, 9));
        DBMS_OUTPUT.PUT_LINE('Grand total ' || LPAD(v_grand, 18));
    END dept_report;

    FUNCTION month_total (
        p_month IN w16_salary.salary_month%TYPE,
        p_dept  IN w16_emp.dept_name%TYPE DEFAULT NULL
    ) RETURN NUMBER
    IS
        v_total NUMBER;
    BEGIN
        SELECT NVL(SUM(s.net_salary), 0) INTO v_total
        FROM   w16_salary s JOIN w16_emp e ON e.emp_id = s.emp_id
        WHERE  s.salary_month = p_month
        AND    (e.dept_name = p_dept OR p_dept IS NULL);
        RETURN v_total;
    END month_total;

END w19_payroll_pkg;
